STEP 6: Test Scale Factor Hypothesis
─────────────────────────────────────

Run this to check if it's a pure scale error:

  python3 << 'PYTHON'
  import json
  import numpy as np
  
  with open("output/embodied_comparison_v2/data/motions/t2m_00000001.json") as f:
      v2_motion = json.load(f)
  with open("output/embodied_t2m_v3/data/motions/t2m_00000001.json") as f:
      v3_motion = json.load(f)
  
  v2_heights = [f['root_pos'][2] for f in v2_motion['frames']]
  v3_heights = [f['root_pos'][2] for f in v3_motion['frames']]
  
  scale_factor = np.mean(v3_heights) / np.mean(v2_heights)
  print(f"Scale factor: {scale_factor:.4f}")
  print(f"Expected: 0.808 (if scale error)")
  
  # Check if all frames follow this scale
  errors = []
  for i in range(len(v2_heights)):
      expected = v2_heights[i] * scale_factor
      actual = v3_heights[i]
      error = abs(expected - actual)
      if error > 0.01:
          errors.append((i, error))
  
  print(f"Frames with >0.01 error: {len(errors)}")
  if len(errors) == 0:
      print("✓ SCALE ERROR HYPOTHESIS CONFIRMED")
  else:
      print("✗ Scale hypothesis fails - something else is wrong")
      print(f"  Errors at frames: {errors[:5]}")
  PYTHON
