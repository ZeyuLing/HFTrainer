  │   ├─ Frame 1: [tx₁, ty₁, tz₁, r₁₁₀, r₁₁₁, ..., r₂₂₅, r₂₂₆]

  └─────────────────────────────────┴────────────────────────────────────┘


CONVERSION PIPELINE
═══════════════════════════════════════════════════════════════════════════════

  NPZ LOAD
    ↓
    data = np.load(npz_path)
    motion_135 = data['motion_135']      (120, 135)
    fps = data['fps']                     30
    
  EXTRACT COMPONENTS
    ↓
    transl = motion_135[:, :3]           (120, 3)
    rot6d  = motion_135[:, 3:]           (120, 132)
    rot6d  = rot6d.reshape(120, 22, 6)   (120, 22, 6)
    
  CONVERT ROTATIONS: rot6d → axis-angle
    ├─ Input:  (120, 22, 6) row-major [R00, R01, R10, R11, R20, R21]
    │
    ├─ Reorder: [R00, R10, R20, R01, R11, R21]  (column-major)
    │
    ├─ Extract: a1 = [R00, R10, R20], a2 = [R01, R11, R21]
    │
    ├─ Gram-Schmidt:
    │   b1 = a1 / ||a1||
    │   b2 = (a2 - (b1·a2)b1) / ||...||
    │   b3 = b1 × b2
    │   R_matrix = [b1, b2, b3]  (3×3)
    │
    └─ Axis-angle: scipy.spatial.transform.Rotation.from_matrix().as_rotvec()
       Output: (120, 22, 3)
       
  DECOMPOSE JOINT POSES
    ↓
    root_orient = aa[:, 0, :]        (120, 3)    ← Joint 0 (pelvis)
    body_pose   = aa[:, 1:22, :]     (120, 21, 3) ← Joints 1-21
    
  BUILD SMPL POSES (example: SMPL+H with 156 dims)
    ↓
    poses_per_frame = zeros((120, 156))
    poses_per_frame[:, :3] = root_orient         (120, 3)
    poses_per_frame[:, 3:66] = body_pose.reshape(120, 63)
    poses_per_frame[:, 66:156] = 0               (zero-padded hands)
    
    Layout (156 dims for SMPL+H):
    ┌─────┬──────────────────┬──────────────────┐
    │Root │ Body Pose (21×3) │ Hands (30×3) = 0 │
    │ 3   │       63         │        90        │
    └─────┴──────────────────┴──────────────────┘
    
  BUILD JSON FRAME STRUCTURE
    ↓
    frames = []
    for t in range(120):
        frame = [{
            "id": 0,
            "gender": "neutral",
            "smpl_type": "smplh",
            "Rh": [[root_orient[t, 0], root_orient[t, 1], root_orient[t, 2]]],
            "Th": [[transl[t, 0], transl[t, 1], transl[t, 2]]],
            "poses": [[poses_per_frame[t, 0], ..., poses_per_frame[t, 155]]],
            "shapes": [[0.0] * 16],
            "mocap_framerate": 30
        }]
        frames.append(frame)
    
  BUILD OUTPUT JSON
    ↓
    output = {
        "type": "frames",
        "fps": 30,
        "frames": frames  (120 frames)
    }
    
  SAVE JSON
    ↓
    with open("output.json", 'w') as f:
        json.dump(output, f, separators=(',', ':'))
    
    File size: ~425 KB (compact format)
