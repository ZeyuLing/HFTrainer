┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃ 1. CORE STEP FUNCTION (the main computation)                                 ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    prev_sample = sample + (sigma_next - sigma) * model_output

    Input Variables:
    ├─ sample        : Current noisy latent x_t [tensor]
    ├─ model_output  : Network prediction (flow direction) m_t [tensor]
    ├─ sigma         : Current noise level σ_t [scalar from sigmas[step_index]]
    └─ sigma_next    : Next noise level σ_{t-1} [scalar from sigmas[step_index+1]]

    Output:
    └─ prev_sample   : Denoised sample x_{t-1} [tensor]

    Mathematical Form:
    ─────────────────
    x_{t-1} = x_t + (σ_{t-1} - σ_t) · m_t
    
    Note: This is a LINEAR EULER STEP in sigma space
          σ_{t-1} < σ_t (moving from noise to clean)
          So (σ_{t-1} - σ_t) is ALWAYS NEGATIVE


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃ 2. SIGMA & TIMESTEP RELATIONSHIP                                             ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    Bidirectional Conversion:
    ────────────────────────
    
    σ = t / num_train_timesteps
    
    t = σ * num_train_timesteps
    
    Example (num_train_timesteps=1000):
    ────────────────────────────────────
    t=1000  ──→  σ=1.000  (max noise)
    t=500   ──→  σ=0.500  (half denoised)
    t=0     ──→  σ=0.000  (min noise/clean)


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃ 3. SET_TIMESTEPS() - INFERENCE SCHEDULE SETUP                                 ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    Called once before inference:
    
    scheduler.set_timesteps(num_inference_steps=50, device='cuda')
    
    What it does:
    ─────────────
    1. Creates linearly-spaced timesteps from 1000 to 0
       timesteps = linspace(1000, 0, 50)
       → [1000, 979.6, 959.2, ..., 20.4, 0]
    
    2. Converts to sigma scale:
       sigmas = timesteps / 1000
       → [1.0, 0.9796, 0.9592, ..., 0.0204, 0.0]
    
    3. Applies shift transform (if not using dynamic shifting):
       σ_shifted = shift * σ / (1 + (shift - 1) * σ)
       With shift=1.0: no change
       With shift>1.0: stretches the schedule
    
    4. Pads with a zero:
       self.sigmas = [σ_0, σ_1, ..., σ_49, 0.0]
                     └─ 50 elements ─┘    └─ padding
       
       Result shape: (51,) instead of (50,)
       
       PURPOSE: During denoising loop, sigmas[step_index + 1] must be valid
                Even for the last step where step_index=49:
                σ_next = sigmas[50] = 0.0 ✓


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃ 4. DENOISING LOOP EXECUTION                                                   ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    Code Flow:
    ──────────
    
    # Before loop: Initialize scheduler
    scheduler.set_timesteps(num_inference_steps=50, device=device)
    timesteps = scheduler.timesteps  # Shape: [50]
    
    # Denoising loop
    for i, t in enumerate(timesteps):
        #                 ↑ t is actual timestep value, NOT index
        #                   (e.g., t=1000, t=979.6, t=959.2, ...)
        
        # First call: initializes step_index
        # scheduler._init_step_index(t) looks up: which index has this timestep?
        
        sigma = scheduler.sigmas[scheduler.step_index]          # e.g., sigmas[0]=1.0
        sigma_next = scheduler.sigmas[scheduler.step_index + 1] # e.g., sigmas[1]=0.9796
        
        # Run model
        model_pred = transformer(latents, timestep=t, ...)
        
        # Scheduler step
        latents = scheduler.step(model_pred, t, latents).prev_sample
        
        # Inside step():
        #   prev_sample = latents + (0.9796 - 1.0) * model_pred
        #                = latents - 0.0204 * model_pred
        #   step_index += 1  (moves to next sigma)
    
    # After loop: latents are clean samples (sigma ≈ 0)


┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃ 5. SHIFT TRANSFORMATIONS                                                      ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    Static Shift (use_dynamic_shifting=False):
    ─────────────────────────────────────────
    σ_shifted = shift * σ / (1 + (shift - 1) * σ)
    
    Default: shift=1.0
    → σ_shifted = 1.0 * σ / (1 + 0) = σ (no change)
    
    With shift=1.15:
    → Stretches early timesteps, compresses later ones
    → More sampling steps devoted to high-noise region


    Dynamic Shift (use_dynamic_shifting=True):
    ──────────────────────────────────────────
    σ_shifted = time_shift(μ, 1.0, σ)
    
    where:
    time_shift(μ, σ, t) = e^μ / (e^μ + (1/t - 1)^σ)
    
    μ: parameter controlling shift magnitude (e.g., log(16/9))
    Adapts schedule based on sequence length

┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃ 7. COMPLETE INFERENCE SEQUENCE EXAMPLE                                        ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    Assuming: num_inference_steps=3 (for brevity)
    
    After set_timesteps():
    ─────────────────────
    timesteps    = [1000.0, 500.0, 0.0]
    sigmas       = [1.0, 0.5, 0.0, 0.0]  ← Note the padding!
                    ↑0   ↑1  ↑2  ↑padding
    
    Iteration 0:
    ───────────
    t = timesteps[0] = 1000.0
    _init_step_index(1000.0) → finds sigmas index → step_index=0
    
    sigma = sigmas[0] = 1.0
    sigma_next = sigmas[1] = 0.5
    prev_sample = latents + (0.5 - 1.0) * model_pred
               = latents - 0.5 * model_pred
    step_index → 1
    
    Iteration 1:
    ───────────
    t = timesteps[1] = 500.0
    step_index already set = 1
    
    sigma = sigmas[1] = 0.5
    sigma_next = sigmas[2] = 0.0
    prev_sample = latents + (0.0 - 0.5) * model_pred
               = latents - 0.5 * model_pred
    step_index → 2
    
    Iteration 2:
    ───────────
    t = timesteps[2] = 0.0
    step_index already set = 2
    
    sigma = sigmas[2] = 0.0
    sigma_next = sigmas[3] = 0.0  ← Padding enables this!
    prev_sample = latents + (0.0 - 0.0) * model_pred
               = latents + 0
               = latents (no change, already clean)
    step_index → 3
    
    Loop ends: latents are now clean samples


╔════════════════════════════════════════════════════════════════════════════════╗
║                              KEY TAKEAWAYS                                    ║
╚════════════════════════════════════════════════════════════════════════════════╝

1. The core equation is SIMPLE: x_new = x_old + Δσ * model_pred
   Δσ = sigma_next - sigma (always negative, moving toward clean)

2. Sigmas range from 1.0 (noisy) to 0.0 (clean)
   Linear relationship with timesteps: σ = t / T

3. Scheduler maintains TWO arrays:
   - timesteps: [num_inference_steps]  ← Used in the loop
   - sigmas:    [num_inference_steps + 1]  ← Step computation (has extra 0 at end)

4. Step index auto-increments: determines which (sigma, sigma_next) pair to use

5. Precision: float32 during computation, then cast back to model.dtype

6. The scheduler is DETERMINISTIC (no randomness unless using s_churn parameters)
